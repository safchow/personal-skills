import { test } from '@playwright/test'
import { bearer } from './helpers/client.js'
import { expectStatus } from './helpers/assertions.js'
import { createAuthedUser } from './helpers/fixtures.js'

/**
 * One endpoint per file (see example-create.spec.ts for the canonical structure).
 *
 * This file: the item delete endpoint. Matrix rows: happy (with side-effect
 * read), idempotency, unknown id.
 */
test.describe('DELETE /api/<resource>/:id', () => {
  test('deletes the resource so a follow-up read 404s', async ({ request }) => {
    // Arrange
    const { token } = await createAuthedUser(request)
    const created = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: 'temp', amount: 5 },
    })
    const id = (await created.json()).resource.id

    // Act
    const res = await request.delete(`/api/<resource>/${id}`, { headers: bearer(token) })

    // Assert: status, then the side effect confirmed through the API.
    await expectStatus(res, 204)
    const read = await request.get(`/api/<resource>/${id}`, { headers: bearer(token) })
    await expectStatus(read, 404)
  })

  test('is idempotent: deleting an already-deleted resource 404s', async ({ request }) => {
    const { token } = await createAuthedUser(request)
    const created = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: 'temp', amount: 5 },
    })
    const id = (await created.json()).resource.id
    await request.delete(`/api/<resource>/${id}`, { headers: bearer(token) })

    const res = await request.delete(`/api/<resource>/${id}`, { headers: bearer(token) })
    await expectStatus(res, 404)
  })

  test('requires authentication (401)', async ({ request }) => {
    const res = await request.delete('/api/<resource>/anything')
    await expectStatus(res, 401)
  })
})
