import { expect, test } from '@playwright/test'
import { bearer } from './helpers/client.js'
import { expectErrorCode, expectStatus } from './helpers/assertions.js'
import { createAuthedUser } from './helpers/fixtures.js'

/**
 * CANONICAL SPEC STRUCTURE — copy this shape for every resource.
 *
 * File: one spec file per resource (or per endpoint for large resources).
 * Suite: ONE describe block per endpoint, named exactly "<METHOD> <path>".
 *        Never mix endpoints in one describe — GET/POST/DELETE each get their own.
 * Test:  one test per coverage-matrix row (SKILL.md), named as a behavior
 *        statement. Each test body follows Arrange / Act / Assert:
 *          Arrange — seed state through fixtures (API calls), namespaced by uniqueId().
 *          Act     — a SINGLE call to the endpoint under test.
 *          Assert  — status code, body shape, and side effects via a follow-up API read.
 *
 * Not every matrix row applies to every endpoint (a collection POST has no 404;
 * an item GET has no 422). Cover the rows that apply to THIS endpoint.
 *
 * Isolation: no beforeEach reset — each test creates its own uniqueId()-namespaced
 * data. Add `await resetIfAvailable(request)` only if the service offers a reset.
 */

// ---------------------------------------------------------------------------
// Collection endpoint: create. Matrix rows: happy, validation, boundary,
// authn, idempotency/side-effects.
// ---------------------------------------------------------------------------
test.describe('POST /api/<resource>', () => {
  test('creates the resource and a follow-up read reflects it', async ({ request }) => {
    // Arrange
    const { token } = await createAuthedUser(request)

    // Act
    const res = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: 'example', amount: 25 },
    })

    // Assert: status, body, then the side effect through the API (never a DB read).
    await expectStatus(res, 201)
    const created = (await res.json()).resource
    expect(created).toMatchObject({ name: 'example', amount: 25 })

    const read = await request.get(`/api/<resource>/${created.id}`, {
      headers: bearer(token),
    })
    await expectStatus(read, 200)
    expect((await read.json()).resource).toMatchObject({ id: created.id })
  })

  test('rejects invalid input with a validation error (422)', async ({ request }) => {
    const { token } = await createAuthedUser(request)

    const res = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: '', amount: -1 },
    })

    await expectStatus(res, 422)
    await expectErrorCode(res, 'E_VALIDATION')
  })

  test('rejects an amount of zero at the lower boundary (422)', async ({ request }) => {
    const { token } = await createAuthedUser(request)

    const res = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: 'edge', amount: 0 },
    })

    await expectStatus(res, 422)
    await expectErrorCode(res, 'E_VALIDATION')
  })

  test('requires authentication (401)', async ({ request }) => {
    const res = await request.post('/api/<resource>', {
      data: { name: 'example', amount: 25 },
    })

    await expectStatus(res, 401)
  })
})

// ---------------------------------------------------------------------------
// Item endpoint: read one. Matrix rows: happy, authn, authz/not-owner,
// unknown id.
// ---------------------------------------------------------------------------
test.describe('GET /api/<resource>/:id', () => {
  test('returns the resource to its owner', async ({ request }) => {
    const { token } = await createAuthedUser(request)
    const created = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: 'owned', amount: 10 },
    })
    const id = (await created.json()).resource.id

    const res = await request.get(`/api/<resource>/${id}`, { headers: bearer(token) })

    await expectStatus(res, 200)
    expect((await res.json()).resource).toMatchObject({ id })
  })

  test('requires authentication (401)', async ({ request }) => {
    const res = await request.get('/api/<resource>/anything')
    await expectStatus(res, 401)
  })

  test('forbids reading another user\u2019s resource (403)', async ({ request }) => {
    const owner = await createAuthedUser(request)
    const created = await request.post('/api/<resource>', {
      headers: bearer(owner.token),
      data: { name: 'owned', amount: 10 },
    })
    const id = (await created.json()).resource.id

    const intruder = await createAuthedUser(request)
    const res = await request.get(`/api/<resource>/${id}`, { headers: bearer(intruder.token) })

    await expectStatus(res, 403)
    await expectErrorCode(res, 'E_FORBIDDEN')
  })

  test('404s for an unknown id', async ({ request }) => {
    const { token } = await createAuthedUser(request)
    const res = await request.get('/api/<resource>/does_not_exist', { headers: bearer(token) })
    await expectStatus(res, 404)
  })
})

// ---------------------------------------------------------------------------
// Item endpoint: delete. Matrix rows: happy, authz, unknown id, state
// conflict, idempotency.
// ---------------------------------------------------------------------------
test.describe('DELETE /api/<resource>/:id', () => {
  test('deletes the resource so a follow-up read 404s', async ({ request }) => {
    const { token } = await createAuthedUser(request)
    const created = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: 'temp', amount: 5 },
    })
    const id = (await created.json()).resource.id

    const res = await request.delete(`/api/<resource>/${id}`, { headers: bearer(token) })
    await expectStatus(res, 204)

    // Side effect confirmed through the API.
    const read = await request.get(`/api/<resource>/${id}`, { headers: bearer(token) })
    await expectStatus(read, 404)
  })

  test('is idempotent: deleting an already-deleted resource 404s', async ({ request }) => {
    const { token } = await createAuthedUser(request)
    const created = await request.post('/api/<resource>', {
      headers: bearer(token),
      data: { name: 'temp', amount: 5 },
    })
    const id = (await created.json()).resource.id
    await request.delete(`/api/<resource>/${id}`, { headers: bearer(token) })

    const res = await request.delete(`/api/<resource>/${id}`, { headers: bearer(token) })
    await expectStatus(res, 404)
  })
})
