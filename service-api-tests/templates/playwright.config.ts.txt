import { defineConfig } from '@playwright/test'

/**
 * Playwright config for a black-box REST API suite.
 *
 *   TEST_BASE_URL=https://staging.example.com npx playwright test
 *
 * The service under test is a black box — start it however the repo already
 * does (dev command, container, deployed instance) and point TEST_BASE_URL at
 * it. This config intentionally knows nothing about the database or framework.

 * Tests run serially (workers: 1) so ordering is deterministic even when the

 */

  use: {

  projects: [{ name: 'api', testMatch: /.*\.spec\.ts$/ }],
